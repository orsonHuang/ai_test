## 2. 核心脚本

### 2.1 TurnQueueManager (行动队列)

```gdscript
class_name TurnQueueManager
extends Node

# 行动队列
var action_queue: Array[BattleUnit] = []
var current_turn_index: int = 0
var is_processing: bool = false

# 构建行动队列（按速度排序）
func build_queue(units: Array[BattleUnit]):
    action_queue.clear()
    # 只添加存活单位
    for unit in units:
        if unit.is_alive:
            action_queue.append(unit)
    # 按速度降序排序
    action_queue.sort_custom(func(a, b): return a.current_stats.speed > b.current_stats.speed)
    current_turn_index = 0

# 处理当前回合
func process_current_turn():
    if current_turn_index >= action_queue.size():
        # 回合结束，开始新回合
        start_new_round()
        return
    
    var current_unit = action_queue[current_turn_index]
    if not current_unit.is_alive:
        # 单位已死亡，跳过
        next_turn()
        return
    
    # 执行单位行动
    await current_unit.execute_turn()
    
    # 进入下一回合
    next_turn()

# 下一回合
func next_turn():
    current_turn_index += 1
    process_current_turn()

# 新回合开始
func start_new_round():
    # 回合开始结算（BUFF、DOT等）
    for unit in action_queue:
        unit.on_round_start()
    
    # 重新构建队列（可能有单位死亡或速度变化）
    build_queue(get_all_units())
    process_current_turn()
```

### 2.2 BattleUnit (战斗单位)

```gdscript
class_name BattleUnit
extends Node2D

# 基础属性
var base_stats: UnitStats
var current_stats: UnitStats

# 状态
var is_alive: bool = true
var hp: int
var shield: int = 0

# 技能
var skills: Array[Skill] = []
var normal_attack: Skill

# BUFF管理
var buff_manager: BuffManager

# AI控制器
var ai_controller: AIController

# 初始化
func init(stats: UnitStats, skill_data: Array):
    base_stats = stats
    current_stats = stats.duplicate()
    hp = current_stats.max_hp
    
    # 初始化技能
    for data in skill_data:
        var skill = SkillFactory.create(data)
        skills.append(skill)
    
    buff_manager = BuffManager.new(self)
    ai_controller = AIController.new(self)

# 执行回合
func execute_turn():
    # AI决策
    var action = ai_controller.decide_action()
    
    match action.type:
        ActionType.SKILL:
            await use_skill(action.skill)
        ActionType.ATTACK:
            await perform_attack()
    
    # 回合结束结算
    on_turn_end()

# 使用技能
func use_skill(skill: Skill) -> bool:
    if not skill.can_use():
        return false
    
    skill.start_cooldown()
    
    # 播放动画
    await play_skill_animation(skill)
    
    # 应用效果
    skill.apply_effects(self, get_target(skill))
    
    return true

# 普通攻击
func perform_attack():
    var target = ai_controller.select_attack_target()
    if target == null:
        return
    
    # 播放攻击动画
    await play_attack_animation()
    
    # 计算伤害
    var damage = calculate_damage(normal_attack, target)
    target.take_damage(damage, self)

# 计算伤害
func calculate_damage(skill: Skill, target: BattleUnit) -> int:
    var damage = current_stats.attack * skill.damage_multiplier
    
    # 暴击判定
    if randf() < current_stats.crit_rate:
        damage *= current_stats.crit_damage
    
    # 防御减伤
    var defense_reduction = target.current_stats.defense / (target.current_stats.defense + 100.0)
    damage *= (1 - defense_reduction)
    
    return int(damage)

# 受到伤害
func take_damage(damage: int, attacker: BattleUnit):
    # 先扣护盾
    if shield > 0:
        var shield_absorb = min(shield, damage)
        shield -= shield_absorb
        damage -= shield_absorb
    
    hp -= damage
    
    # 显示伤害数字
    show_damage_number(damage)
    
    # 反击判定
    check_counter_attack(attacker)
    
    if hp <= 0:
        die()

# 回合开始
func on_round_start():
    buff_manager.on_round_start()
    
    # 技能CD减1
    for skill in skills:
        skill.tick_cooldown()

# 回合结束
func on_turn_end():
    buff_manager.on_turn_end()
```

### 2.3 AIController (AI控制器)

```gdscript
class_name AIController
extends RefCounted

var owner: BattleUnit

func _init(unit: BattleUnit):
    owner = unit

# 决定行动
func decide_action() -> Dictionary:
    # 检查主动技能
    for skill in owner.skills:
        if skill.is_active and skill.can_use():
            if check_skill_condition(skill):
                return {"type": ActionType.SKILL, "skill": skill}
    
    # 默认普通攻击
    return {"type": ActionType.ATTACK}

# 检查技能释放条件
func check_skill_condition(skill: Skill) -> bool:
    match skill.ai_condition:
        SkillCondition.ENEMY_CHARGING:
            return is_enemy_charging()
        SkillCondition.BOSS_BREAK:
            return is_boss_in_break_state()
        SkillCondition.SELF_LOW_HP:
            return owner.hp < owner.current_stats.max_hp * 0.3
        SkillCondition.AOE_BENEFIT:
            return get_enemy_count() >= 3
        _:
            return true

# 选择攻击目标
func select_attack_target() -> BattleUnit:
    var enemies = get_enemies()
    if enemies.is_empty():
        return null
    
    # 优先攻击前排
    enemies.sort_custom(func(a, b): return a.position_priority < b.position_priority)
    return enemies[0]
```

## 5. 关键数值设计

### 5.1 速度基准

| 单位类型 | 速度范围 | 说明 |
|----------|----------|------|
| 刺客类 | 120-150 | 先手优势 |
| 战士类 | 90-110 | 标准速度 |
| 法师类 | 80-100 | 较慢但爆发高 |
| 坦克类 | 60-80 | 慢速高防 |
| 普通敌人 | 70-90 | 略低于玩家 |
| BOSS | 50-120 | 依类型而定 |

### 5.2 回合时间估算

假设标准速度 = 100，每回合平均耗时：
- 动画播放：0.5-1 秒
- 伤害数字显示：0.3 秒
- 状态更新：0.1 秒

一场 10 回合的战斗约 20-30 秒（含倍速）。
